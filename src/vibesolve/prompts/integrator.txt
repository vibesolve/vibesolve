## Mandatory REST contract

Create rest/SolverResource.java at @Path("api"), with all SEVEN named endpoints:
GET /all, GET /generate, POST /solve, GET /solution/{jobId},
GET /status/{jobId}, DELETE /stop/{jobId}, PUT /analyze.
solve returns a job ID asynchronously, not a solved object. Every endpoint has
its own subpath. /generate JSON must be directly reusable as /solve input.

Adapt PACKAGE, SOLUTION_CLASS and the DataGenerator import/call to the existing
manifest. Import the actual solution class from its domain package. Keep
SolverStatus visible on the solution. Preserve correct withBestSolutionConsumer
or withBestSolutionEventConsumer callbacks; both exist in 1.31.0. Event callbacks
receive event.solution(). The template uses computeIfPresent so late callbacks
cannot recreate explicitly removed jobs. SolverManager.SolverStatus,
SolverManagerFactory, getSolverJob and solveAsync are not this API.

~~~java
package {{PACKAGE}}.rest;

import ai.timefold.solver.core.api.score.analysis.ScoreAnalysis;
import ai.timefold.solver.core.api.score.buildin.hardsoft.HardSoftScore;
import ai.timefold.solver.core.api.solver.ScoreAnalysisFetchPolicy;
import ai.timefold.solver.core.api.solver.SolutionManager;
import ai.timefold.solver.core.api.solver.SolverManager;
import ai.timefold.solver.core.api.solver.SolverStatus;
import {{PACKAGE}}.generator.DataGenerator;
import {{PACKAGE}}.domain.{{SOLUTION_CLASS}};
import jakarta.inject.Inject;
import jakarta.ws.rs.Consumes;
import jakarta.ws.rs.DELETE;
import jakarta.ws.rs.GET;
import jakarta.ws.rs.POST;
import jakarta.ws.rs.PUT;
import jakarta.ws.rs.Path;
import jakarta.ws.rs.PathParam;
import jakarta.ws.rs.Produces;
import jakarta.ws.rs.QueryParam;
import jakarta.ws.rs.WebApplicationException;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.ArrayList;
import java.util.Collection;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ConcurrentMap;

@Path("api")
public class SolverResource {

    private static final Logger LOGGER = LoggerFactory.getLogger(SolverResource.class);

    private final SolverManager<{{SOLUTION_CLASS}}, String> solverManager;
    private final SolutionManager<{{SOLUTION_CLASS}}, HardSoftScore> solutionManager;

    // Completed jobs remain until explicitly stopped/removed.
    private final ConcurrentMap<String, Job> jobIdToJob = new ConcurrentHashMap<>();

    // Required for CDI proxy construction; application calls use the injected constructor.
    public SolverResource() {
        this.solverManager = null;
        this.solutionManager = null;
    }

    @Inject
    public SolverResource(SolverManager<{{SOLUTION_CLASS}}, String> solverManager,
            SolutionManager<{{SOLUTION_CLASS}}, HardSoftScore> solutionManager) {
        this.solverManager = solverManager;
        this.solutionManager = solutionManager;
    }

    @GET
    @Path("all")
    @Produces(MediaType.APPLICATION_JSON)
    public Collection<String> listAll() {
        return new ArrayList<>(jobIdToJob.keySet());
    }

    @GET
    @Path("generate")
    @Produces(MediaType.APPLICATION_JSON)
    public {{SOLUTION_CLASS}} generate() {
        return new DataGenerator().generate();
    }

    @POST
    @Path("solve")
    @Consumes(MediaType.APPLICATION_JSON)
    @Produces(MediaType.TEXT_PLAIN)
    public String solve({{SOLUTION_CLASS}} problem) {
        String jobId = UUID.randomUUID().toString();
        jobIdToJob.put(jobId, Job.ofSolution(problem));
        solverManager.solveBuilder()
                .withProblemId(jobId)
                .withProblem(problem)
                .withBestSolutionEventConsumer(event -> jobIdToJob.computeIfPresent(
                        jobId, (id, previous) -> Job.ofSolution(event.solution())))
                .withExceptionHandler((jobId_, exception) -> {
                    jobIdToJob.computeIfPresent(jobId, (id, previous) -> Job.ofException(exception));
                    LOGGER.error("Failed solving jobId ({}).", jobId, exception);
                })
                .run();
        return jobId;
    }

    @GET
    @Path("solution/{jobId}")
    @Produces(MediaType.APPLICATION_JSON)
    public {{SOLUTION_CLASS}} getBestSolution(@PathParam("jobId") String jobId) {
        {{SOLUTION_CLASS}} solution = getSolutionAndCheckForExceptions(jobId);
        solution.setSolverStatus(solverManager.getSolverStatus(jobId));
        return solution;
    }

    @GET
    @Path("status/{jobId}")
    @Produces(MediaType.APPLICATION_JSON)
    public SolverStatus getStatus(@PathParam("jobId") String jobId) {
        getSolutionAndCheckForExceptions(jobId); // validates job exists and re-throws solver exceptions
        return solverManager.getSolverStatus(jobId);
    }

    @DELETE
    @Path("stop/{jobId}")
    @Produces(MediaType.APPLICATION_JSON)
    public {{SOLUTION_CLASS}} stopSolving(@PathParam("jobId") String jobId) {
        solverManager.terminateEarly(jobId);
        {{SOLUTION_CLASS}} solution = getBestSolution(jobId); // fetch before removal; throws 404 if missing
        jobIdToJob.remove(jobId); // graceful cleanup — no-op if already absent
        return solution;
    }

    @PUT
    @Path("analyze")
    @Consumes(MediaType.APPLICATION_JSON)
    @Produces(MediaType.APPLICATION_JSON)
    public ScoreAnalysis<HardSoftScore> analyze({{SOLUTION_CLASS}} problem,
            @QueryParam("fetchPolicy") ScoreAnalysisFetchPolicy fetchPolicy) {
        return fetchPolicy == null
                ? solutionManager.analyze(problem)
                : solutionManager.analyze(problem, fetchPolicy);
    }

    private {{SOLUTION_CLASS}} getSolutionAndCheckForExceptions(String jobId) {
        Job job = jobIdToJob.get(jobId);
        if (job == null) {
            throw new WebApplicationException("No solution found for jobId: " + jobId,
                    Response.Status.NOT_FOUND);
        }
        if (job.exception != null) {
            throw new WebApplicationException(job.exception.getMessage(),
                    Response.Status.INTERNAL_SERVER_ERROR);
        }
        return job.solution;
    }

    private record Job({{SOLUTION_CLASS}} solution, Throwable exception) {
        static Job ofSolution({{SOLUTION_CLASS}} solution) { return new Job(solution, null); }
        static Job ofException(Throwable e) { return new Job(null, e); }
    }
}
~~~

## Solver and constraint tests

Generate SolverIntegrationTest. For five small independently generated datasets:
- Prefer generate(seed) with fixed seeds if the existing generator supports it;
  otherwise call its existing method. Do not modify the generator here.
- Before solving, assert all genuine basic variables are null or all planning
  lists empty. Use fresh graphs, not the same solved instance repeatedly.
- Load solverConfigTest.xml, build a Solver, and solve for its configured 15
  seconds in FULL_ASSERT. Assert non-null initialized HardSoftScore AND that the
  solver actually assigned at least one basic variable/list element.
- Check feasibility only for deliberately feasible samples; do not weaken a
  real constraint or preassign variables to make these assertions pass.
- If fixed relationship facts reference entities, assertSame those references
  against canonical entities in the SOLVED solution, not just the input graph.
  This catches missing planning-clone support that JSON-only checks cannot fix.
- Round-trip an existing solved result's exact bytes through the standalone IO
  class's write/read methods (streams or @TempDir). Assert a non-null score and
  exact HardSoftScore equality. No clearing scores, test-only ObjectMapper, REST
  substitute or swallowed errors. Reuse an existing solve.
- Keep data small so five solves plus REST testing fit the 180-second validator
  test budget. Avoid repeated application startup or additional long solves.

Also add focused ConstraintVerifier tests for important requirements. Cover a
satisfying case, a violation, equality boundaries, zero assignments, and the
actual objective direction/priorities where applicable. For max-N, test N and
N+1. For minimum coverage include resources with zero assignments. Test makespan
against examples distinguishing maximum from sum. For finish-by routing windows,
include early arrival with waiting. Do not only assert that a score exists.
For ALL-of requirements, include a two-value requirement and a candidate with
only one of those values; it must be rejected. Do not reduce the domain to a
single required value merely because simpler fixtures would pass.
Constructors/accessors must match the manifest. Register all genuine/shadow
entity classes when constructing ConstraintVerifier. Ordinary given(...) does
not initialize shadows: set them explicitly in the fixture when needed.
Full-solution score consistency is not an independent business-rule oracle.

ConstraintVerifier has two DIFFERENT assertion APIs. Do not mix them:
- Import ai.timefold.solver.test.api.score.stream.ConstraintVerifier. Its generic
  parameters are <Provider, Solution>, NOT <Provider, HardSoftScore>. Initialize
  with ConstraintVerifier.build(new Provider(), Solution.class, Entity.class),
  listing every genuine/shadow entity class when there is more than one.
- Single rule: verifier.verifyThat(Provider::rule).given(facts...).penalizesBy(1)
  or .penalizesBy(0). This asserts match weight, not the weighted score.
  rewardsWith(...) is the corresponding reward assertion. There is NO scores()
  method on SingleConstraintAssertion.
- Whole provider: verifier.verifyThat().given(facts...).scores(HardSoftScore.of(-1, 0)).
  Only this no-selector verifyThat() form supports an exact total score assertion.
- The single-rule selector takes (provider, factory). Provider::rule works for
  an accessible INSTANCE method rule(ConstraintFactory). For a static rule use
  (provider, factory) -> Provider.rule(factory), not Provider::rule. Inspect the
  existing method declaration. Do not change a correct provider just for style.
- ConstraintFactory owns forEachUniquePair: call factory.forEachUniquePair(...)
  before filtering, not a forEachUniquePair method on an existing stream.

## Mandatory native JSON REST test

Generate SolverResourceTest.java from this single test. No alternate
synchronous endpoint example. This test uses exact response bytes, not a Java
object reserialized by REST-assured. Do not weaken its assertions. The score
comparison verifies native round-trip consistency, not business-rule correctness.

~~~java
package {{PACKAGE}}.rest;

import ai.timefold.solver.core.api.score.buildin.hardsoft.HardSoftScore;
import io.quarkus.test.junit.QuarkusTest;
import io.restassured.http.ContentType;
import io.restassured.path.json.JsonPath;
import org.junit.jupiter.api.Test;
import java.nio.charset.StandardCharsets;
import static io.restassured.RestAssured.given;
import static org.junit.jupiter.api.Assertions.*;

@QuarkusTest
class SolverResourceTest {
    private static final String BASE = "/api";

    @Test
    void testAllSevenEndpoints() throws InterruptedException {
        byte[] generatedJson = given().when().get(BASE + "/generate")
                .then().statusCode(200).extract().asByteArray();
        String jobId = given().contentType(ContentType.JSON).body(generatedJson)
                .when().post(BASE + "/solve")
                .then().statusCode(200).extract().asString();
        assertFalse(jobId.isBlank());
        try {
            assertTrue(given().when().get(BASE + "/all").then().statusCode(200)
                    .extract().jsonPath().getList("$", String.class).contains(jobId));
            boolean completed = false;
            for (int i = 0; i < 45; i++) {
                String status = given().when().get(BASE + "/status/" + jobId)
                        .then().statusCode(200).extract().jsonPath().getString("$");
                if ("NOT_SOLVING".equals(status)) {
                    completed = true;
                    break;
                }
                Thread.sleep(1000);
            }
            assertTrue(completed, "Solver did not finish within 45 seconds");
            byte[] solutionJson = given().when().get(BASE + "/solution/" + jobId)
                    .then().statusCode(200).extract().asByteArray();
            String solvedScore = JsonPath.from(new String(solutionJson,
                    StandardCharsets.UTF_8)).getString("score");
            assertNotNull(solvedScore);
            HardSoftScore score = HardSoftScore.parseScore(solvedScore);
            assertTrue(score.isSolutionInitialized());
            String analyzedScore = given().contentType(ContentType.JSON)
                    .body(solutionJson).when().put(BASE + "/analyze")
                    .then().statusCode(200).extract().jsonPath().getString("score");
            assertEquals(score, HardSoftScore.parseScore(analyzedScore));
        } finally {
            given().when().delete(BASE + "/stop/" + jobId).then().statusCode(200);
        }
        assertFalse(given().when().get(BASE + "/all").then().statusCode(200)
                .extract().jsonPath().getList("$", String.class).contains(jobId));
        given().when().get(BASE + "/solution/" + jobId).then().statusCode(404);
        given().when().get(BASE + "/solution/nonexistent-job-id-00000")
                .then().statusCode(404);
    }
}
~~~
